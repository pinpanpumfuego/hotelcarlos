# 'production' oculta los errores técnicos al visitante y apaga la barra
# de depuración. En 'development' se enseñan rutas y consultas: en un
# servidor público eso es regalar información a quien quiera atacarlo.
CI_ENVIRONMENT = production

# ── Dirección de la web ──────────────────────────────────────────────
# Con https:// y con la barra final. Tiene que ser exacta o los enlaces
# y el calendario de reservas apuntarán a donde no es.
app.baseURL = 'https://<tu-dominio>/'

app.indexPage = ''

# Obliga a que todo vaya por https. Sin esto no funcionan: instalar el
# comandero como app, abrirlo sin señal, la cámara y el GPS del fichaje,
# ni el «oído cocina».
app.forceGlobalSecureRequests = true

app.appTimezone = 'America/Bogota'
app.defaultLocale = 'es'
